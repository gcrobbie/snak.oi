// ── DB ────────────────────────────────────────────────────────────────────────
async function dbRead() {
  const r = await fetch(JSONBIN_URL + '/latest', { headers:{ 'X-Master-Key': JSONBIN_API_KEY } });
  if (!r.ok) throw new Error('DB read failed');
  return (await r.json()).record;
}
async function dbWrite(data) {
  const r = await fetch(JSONBIN_URL, {
    method:'PUT',
    headers:{ 'Content-Type':'application/json', 'X-Master-Key': JSONBIN_API_KEY },
    body: JSON.stringify(data)
  });
  if (!r.ok) throw new Error('DB write failed');
  return r.json();
}
async function hashPw(pw) {
  const buf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(pw + 'snakoi_salt'));
  return Array.from(new Uint8Array(buf)).map(b => b.toString(16).padStart(2,'0')).join('');
}

// ── AUTH ──────────────────────────────────────────────────────────────────────
function toggleAuthMode() {
  authMode = authMode === 'signin' ? 'register' : 'signin';
  document.getElementById('auth-title').textContent      = authMode === 'signin' ? 'Sign In' : 'Create Account';
  document.getElementById('auth-submit-btn').textContent = authMode === 'signin' ? 'Sign In' : 'Register';
  document.getElementById('auth-toggle').innerHTML = authMode === 'signin'
    ? 'No account? <span onclick="toggleAuthMode()">Create one</span>'
    : 'Already have one? <span onclick="toggleAuthMode()">Sign in</span>';
  document.getElementById('auth-error').textContent = '';
}

async function authSubmit() {
  const username = document.getElementById('auth-username').value.trim();
  const password = document.getElementById('auth-password').value;
  const errEl    = document.getElementById('auth-error');
  const btn      = document.getElementById('auth-submit-btn');
  errEl.textContent = '';
  if (!username || !password) { errEl.textContent = 'Please fill in all fields.'; return; }
  if (username.length < 2)    { errEl.textContent = 'Username must be 2+ chars.'; return; }
  btn.innerHTML = '<span class="spinner"></span>Loading…'; btn.disabled = true;
  try {
    const db   = await dbRead();
    const hash = await hashPw(password);
    if (authMode === 'register') {
      if (db.users[username]) { errEl.textContent = 'Username taken.'; return; }
      db.users[username] = { hash, best:0, skin:'classic' };
      await dbWrite(db);
      toast('Account created! 🎉', 'success');
    } else {
      const user = db.users[username];
      if (!user || user.hash !== hash) { errEl.textContent = 'Invalid username or password.'; return; }
    }
    currentUser = username;
    localStorage.setItem('snakoi_user', username);
    localStorage.setItem('snakoi_hash', hash);
    showMenu();
  } catch(e) {
    errEl.textContent = 'Connection error. Check your JSONBin config.';
  } finally {
    btn.innerHTML = authMode === 'signin' ? 'Sign In' : 'Register'; btn.disabled = false;
  }
}

async function autoLogin() {
  const savedUser = localStorage.getItem('snakoi_user');
  const savedHash = localStorage.getItem('snakoi_hash');
  if (!savedUser || !savedHash) return false;
  try {
    const db   = await dbRead();
    const user = db.users[savedUser];
    if (user && user.hash === savedHash) { currentUser = savedUser; return true; }
  } catch(e) {}
  localStorage.removeItem('snakoi_user');
  localStorage.removeItem('snakoi_hash');
  return false;
}

function logout() {
  currentUser = null;
  localStorage.removeItem('snakoi_user');
  localStorage.removeItem('snakoi_hash');
  showAuth();
}

function renderSkinSelector(best) {
  const container = document.getElementById('skin-selector');
  container.innerHTML = '';
  SKINS.forEach(skin => {
    const unlocked = best >= skin.unlockAt;
    const btn = document.createElement('div');
    btn.className = 'skin-btn' + (skin.id === selectedSkin ? ' selected' : '') + (unlocked ? '' : ' locked');
    btn.title     = unlocked ? skin.name : `${skin.name} — unlock at score ${skin.unlockAt}`;
    btn.innerHTML = skin.label + (unlocked ? '' : '<span class="lock-icon">🔒</span>');
    if (unlocked) btn.onclick = () => {
      selectedSkin = skin.id;
      document.querySelectorAll('.skin-btn').forEach(b => b.classList.remove('selected'));
      btn.classList.add('selected');
    };
    container.appendChild(btn);
  });
  const next = SKINS.find(s => best < s.unlockAt);
  document.getElementById('skin-hint').textContent = next
    ? `Next unlock: ${next.name} at ${next.unlockAt} pts`
    : '🎉 All skins unlocked!';
}

async function loadLeaderboard() {
  document.getElementById('leaderboard-container').innerHTML =
    '<div style="text-align:center;color:var(--muted);padding:14px"><span class="spinner"></span>Loading...</div>';
  try { renderLeaderboard(await dbRead()); }
  catch(e) {
    document.getElementById('leaderboard-container').innerHTML =
      '<div style="color:var(--danger);text-align:center;padding:14px">Failed to load</div>';
  }
}

function renderLeaderboard(db) {
  const entries = Object.entries(db.users || {})
    .map(([name,data]) => ({ name, best:data.best||0 }))
    .sort((a,b) => b.best - a.best).slice(0,10);
  if (!entries.length) {
    document.getElementById('leaderboard-container').innerHTML =
      '<div style="color:var(--muted);text-align:center;padding:14px">No scores yet!</div>';
    return;
  }
  const medals = ['🥇','🥈','🥉'];
  let html = '<table class="leaderboard-table"><thead><tr><th>#</th><th>Player</th><th>Best</th></tr></thead><tbody>';
  entries.forEach((e,i) => {
    const cls = e.name === currentUser ? 'me' : '';
    html += `<tr><td>${medals[i]||i+1}</td><td class="${cls}">${escHtml(e.name)}</td><td>${e.best}</td></tr>`;
  });
  html += '</tbody></table>';
  document.getElementById('leaderboard-container').innerHTML = html;
}

function escHtml(s) { return s.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }

// ── GAME ENGINE ───────────────────────────────────────────────────────────────
const COLS = 20, ROWS = 20;
let canvas, ctx, CELL;
let snake, dir, nextDir, food, powerupItems, particles;
let score, localBest, gameLoop, timerInterval;
let gameRunning  = false;
let activePowerups = {};
let tickInterval = 120;
let shrinkLevel  = 0;
let timeLeft     = 60;
let currentMode  = 'classic';
let currentSkinObj = SKINS[0];
let shrinkTimeout = null;

function getSkinById(id) { return SKINS.find(s => s.id === id) || SKINS[0]; }

function startGame() {
  currentMode    = selectedMode;
  currentSkinObj = getSkinById(selectedSkin);
  showGame();
  canvas = document.getElementById('gameCanvas');
  const size = Math.min(window.innerWidth - 32, 600);
  canvas.width = canvas.height = size;
  CELL = size / COLS;
  ctx  = canvas.getContext('2d');
  document.getElementById('mode-badge').textContent = {
    classic:'🐍 Classic', wraparound:'🌀 No Walls',
    timeattack:'⏱️ Time Attack', shrinking:'📦 Shrinking'
  }[currentMode];
  document.getElementById('hud-timer-box').style.display = currentMode === 'timeattack' ? '' : 'none';
  initGame();
}

function initGame() {
  snake          = [{ x:10,y:10 },{ x:9,y:10 },{ x:8,y:10 }];
  dir            = { x:1, y:0 };
  nextDir        = { x:1, y:0 };
  score          = 0;
  shrinkLevel    = 0;
  timeLeft       = 60;
  tickInterval   = 120;
  gameRunning    = true;
  activePowerups = {};
  powerupItems   = [];
  particles      = [];
  localBest      = parseInt(document.getElementById('hud-best').textContent) || 0;

  document.getElementById('gameover-overlay').classList.remove('active');
  document.getElementById('shrink-warn').textContent = '';
  document.getElementById('shrink-warn').classList.remove('active');

  placeFood();
  updateHUD();
  renderPowerupHUD();

  clearInterval(gameLoop);
  clearInterval(timerInterval);
  clearTimeout(shrinkTimeout);

  gameLoop = setInterval(tick, tickInterval);

  if (currentMode === 'timeattack') {
    document.getElementById('hud-timer').textContent = 60;
    timerInterval = setInterval(() => {
      timeLeft--;
      document.getElementById('hud-timer').textContent = timeLeft;
      if (timeLeft <= 10) document.getElementById('hud-timer-box').style.borderColor = 'var(--danger)';
      if (timeLeft <= 0) { clearInterval(timerInterval); endGame(true); }
    }, 1000);
  }

  if (currentMode === 'shrinking') scheduleShrink();
}

function scheduleShrink() {
  shrinkTimeout = setTimeout(() => {
    if (!gameRunning) return;
    shrinkLevel++;
    const warn = document.getElementById('shrink-warn');
    warn.textContent = `⚠️ Arena shrinking! (level ${shrinkLevel})`;
    warn.classList.add('active');
    setTimeout(() => { warn.classList.remove('active'); warn.textContent = ''; }, 2500);
    // Kill snake if out of new bounds
    const min = shrinkLevel, max = COLS - 1 - shrinkLevel;
    if (snake[0].x < min || snake[0].x > max || snake[0].y < min || snake[0].y > max) {
      endGame(); return;
    }
    if (shrinkLevel < 8) scheduleShrink();
  }, 10000);
}

function placeFood() {
  const min = shrinkLevel, max = COLS - 1 - shrinkLevel;
  let pos, tries = 0;
  do {
    pos = { x: min + Math.floor(Math.random()*(max-min+1)), y: min + Math.floor(Math.random()*(max-min+1)) };
    tries++;
  } while (tries < 30 && snake.some(s => s.x===pos.x && s.y===pos.y));
  food = pos;
}

function maybePlacePowerup() {
  if (powerupItems.length >= 2 || Math.random() > 0.28) return;
  const min = shrinkLevel, max = COLS - 1 - shrinkLevel;
  const type = POWERUP_TYPES[Math.floor(Math.random() * POWERUP_TYPES.length)];
  let pos, tries = 0;
  do {
    pos = { x: min + Math.floor(Math.random()*(max-min+1)), y: min + Math.floor(Math.random()*(max-min+1)) };
    tries++;
  } while (tries < 20 && (
    snake.some(s=>s.x===pos.x&&s.y===pos.y) ||
    (food.x===pos.x&&food.y===pos.y) ||
    powerupItems.some(p=>p.x===pos.x&&p.y===pos.y)
  ));
  powerupItems.push({ ...type, x:pos.x, y:pos.y, spawnedAt:Date.now() });
}

function tick() {
  if (!gameRunning) return;

  const now = Date.now();

  // Expire power-ups
  for (const key of Object.keys(activePowerups)) {
    if (now > activePowerups[key].expiresAt) delete activePowerups[key];
  }

  // Expire old power-up items (15s)
  powerupItems = powerupItems.filter(p => now - p.spawnedAt < 15000);

  // Speed power-up: dynamically change interval
  const wantInterval = activePowerups.speed ? 65 : 120;
  if (tickInterval !== wantInterval) {
    tickInterval = wantInterval;
    clearInterval(gameLoop);
    gameLoop = setInterval(tick, tickInterval);
    return;
  }

  dir = { ...nextDir };
  const head = { x: snake[0].x + dir.x, y: snake[0].y + dir.y };

  // Wall handling
  if (currentMode === 'wraparound') {
    head.x = ((head.x % COLS) + COLS) % COLS;
    head.y = ((head.y % ROWS) + ROWS) % ROWS;
  } else {
    const min = currentMode === 'shrinking' ? shrinkLevel : 0;
    const max = currentMode === 'shrinking' ? COLS-1-shrinkLevel : COLS-1;
    if (head.x < min || head.x > max || head.y < min || head.y > max) {
      if (activePowerups.shield) {
        delete activePowerups.shield;
        toast('🛡️ Shield absorbed a wall hit!', 'info');
        renderPowerupHUD();
        return;
      }
      endGame(); return;
    }
  }

  // Self collision
  if (snake.some(s => s.x===head.x && s.y===head.y)) {
    if (activePowerups.shield) {
      delete activePowerups.shield;
      toast('🛡️ Shield absorbed a self hit!', 'info');
      renderPowerupHUD();
      return;
    }
    endGame(); return;
  }

  snake.unshift(head);

  // Eat food
  if (head.x===food.x && head.y===food.y) {
    const pts = activePowerups.multi ? 2 : 1;
    score += pts;
    updateHUD();
    spawnParticles(food.x, food.y, activePowerups.multi ? '#ff44ff' : currentSkinObj.head, 14);
    placeFood();
    maybePlacePowerup();
  } else {
    snake.pop();
  }

  // Collect power-up
  const pwIdx = powerupItems.findIndex(p => p.x===head.x && p.y===head.y);
  if (pwIdx !== -1) {
    const pw = powerupItems.splice(pwIdx, 1)[0];
    activePowerups[pw.id] = { expiresAt: now + pw.dur, dur: pw.dur };
    toast(`${pw.label} ${pw.desc} activated!`, 'info');
    spawnParticles(head.x, head.y, pw.color, 20);
    renderPowerupHUD();
  }

  renderPowerupHUD();
  draw();
}

// ── PARTICLES ─────────────────────────────────────────────────────────────────
function spawnParticles(gx, gy, color, count = 12) {
  for (let i = 0; i < count; i++) {
    const angle = (Math.PI * 2 * i) / count + Math.random() * 0.3;
    const spd   = 1.5 + Math.random() * 2.5;
    particles.push({
      x: (gx + .5) * CELL, y: (gy + .5) * CELL,
      vx: Math.cos(angle) * spd, vy: Math.sin(angle) * spd,
      alpha: 1, size: 2.5 + Math.random() * 3, color,
    });
  }
}

function updateParticles() {
  particles = particles.filter(p => p.alpha > 0.02);
  for (const p of particles) {
    p.x += p.vx; p.y += p.vy;
    p.vx *= .90; p.vy *= .90;
    p.alpha *= .87;
  }
}

// ── DRAW ──────────────────────────────────────────────────────────────────────
function draw() {
  ctx.fillStyle = '#080810';
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  // Shrink zone tinting
  if (currentMode === 'shrinking' && shrinkLevel > 0) {
    const pad  = shrinkLevel * CELL;
    const inner = canvas.width - pad * 2;
    ctx.fillStyle = 'rgba(255,68,68,0.05)';
    ctx.fillRect(0,0,canvas.width,pad);
    ctx.fillRect(0,canvas.height-pad,canvas.width,pad);
    ctx.fillRect(0,pad,pad,inner);
    ctx.fillRect(canvas.width-pad,pad,pad,inner);
    ctx.strokeStyle = 'rgba(255,68,68,0.4)';
    ctx.lineWidth = 2;
    ctx.strokeRect(pad,pad,inner,inner);
  }

  // Grid
  ctx.strokeStyle = '#0f0f1e'; ctx.lineWidth = .5;
  const gMin = currentMode === 'shrinking' ? shrinkLevel : 0;
  const gMax = currentMode === 'shrinking' ? COLS - shrinkLevel : COLS;
  for (let i = gMin; i <= gMax; i++) {
    ctx.beginPath(); ctx.moveTo(i*CELL,gMin*CELL); ctx.lineTo(i*CELL,gMax*CELL); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(gMin*CELL,i*CELL); ctx.lineTo(gMax*CELL,i*CELL); ctx.stroke();
  }

  // Food
  ctx.fillStyle = 'rgba(255,68,68,0.22)';
  ctx.beginPath(); ctx.arc((food.x+.5)*CELL,(food.y+.5)*CELL,CELL*.75,0,Math.PI*2); ctx.fill();
  ctx.fillStyle = '#ff4444';
  ctx.beginPath(); ctx.roundRect(food.x*CELL+2,food.y*CELL+2,CELL-4,CELL-4,4); ctx.fill();
  ctx.fillStyle = '#884422'; ctx.fillRect((food.x+.5)*CELL-1,food.y*CELL,2,4);

  // Power-up items
  const now = Date.now();
  ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
  powerupItems.forEach(pw => {
    const age = (now - pw.spawnedAt) / 15000;
    ctx.globalAlpha = age > .7 ? 0.35 + 0.35 * Math.sin(now / 100) : 1;
    ctx.fillStyle = pw.color + '30';
    ctx.beginPath(); ctx.arc((pw.x+.5)*CELL,(pw.y+.5)*CELL,CELL*.78,0,Math.PI*2); ctx.fill();
    ctx.globalAlpha = ctx.globalAlpha;
    ctx.font = `${CELL*.65}px serif`;
    ctx.fillText(pw.label, (pw.x+.5)*CELL, (pw.y+.5)*CELL);
    ctx.globalAlpha = 1;
  });

  // Snake
  const hasShield = !!activePowerups.shield;
  snake.forEach((seg, i) => {
    if (i === 0) {
      ctx.fillStyle  = currentSkinObj.head;
      ctx.shadowColor = hasShield ? '#00aaff' : currentSkinObj.head;
      ctx.shadowBlur  = hasShield ? 16 : 9;
    } else {
      // Rainbow skin cycles body colors
      if (currentSkinObj.id === 'rainbow') {
        const hue = (i * 30) % 360;
        ctx.fillStyle = `hsl(${hue},100%,55%)`;
      } else {
        ctx.fillStyle = currentSkinObj.body[Math.min(i-1, currentSkinObj.body.length-1)];
      }
      ctx.shadowBlur = 0;
    }
    const pad = i === 0 ? 1 : 2;
    ctx.beginPath();
    ctx.roundRect(seg.x*CELL+pad, seg.y*CELL+pad, CELL-pad*2, CELL-pad*2, i===0?6:4);
    ctx.fill();
    ctx.shadowBlur = 0;
  });

  // Eyes on head
  const h = snake[0];
  ctx.fillStyle = '#000';
  const ex = dir.x===0?[.3,.7]:(dir.x>0?[.7,.7]:[.3,.3]);
  const ey = dir.y===0?[.3,.7]:(dir.y>0?[.7,.7]:[.3,.3]);
  ctx.beginPath(); ctx.arc((h.x+ex[0])*CELL,(h.y+ey[0])*CELL,2.5,0,Math.PI*2); ctx.fill();
  ctx.beginPath(); ctx.arc((h.x+ex[1])*CELL,(h.y+ey[1])*CELL,2.5,0,Math.PI*2); ctx.fill();

  // Particles
  updateParticles();
  particles.forEach(p => {
    ctx.save();
    ctx.globalAlpha = p.alpha;
    ctx.fillStyle   = p.color;
    ctx.beginPath(); ctx.arc(p.x, p.y, p.size, 0, Math.PI*2); ctx.fill();
    ctx.restore();
  });
}

// ── POWER-UP HUD ──────────────────────────────────────────────────────────────
function renderPowerupHUD() {
  const hud = document.getElementById('powerup-hud');
  hud.innerHTML = '';
  const now = Date.now();
  POWERUP_TYPES.forEach(pw => {
    const active = activePowerups[pw.id];
    if (!active) return;
    const pct = Math.max(0, (active.expiresAt - now) / pw.dur * 100);
    const pill = document.createElement('div');
    pill.className = `pw-pill ${pw.id}`;
    pill.innerHTML = `${pw.label} ${pw.desc} <div class="pw-bar"><div class="pw-bar-fill" style="width:${pct}%"></div></div>`;
    hud.appendChild(pill);
  });
}

// ── END GAME ──────────────────────────────────────────────────────────────────
async function endGame(timeUp = false) {
  gameRunning = false;
  clearInterval(gameLoop);
  clearInterval(timerInterval);
  clearTimeout(shrinkTimeout);
  document.getElementById('go-title').textContent    = timeUp ? "⏱️ Time's Up!" : '💀 Game Over';
  document.getElementById('go-score').textContent    = score;
  document.getElementById('go-best-line').textContent = score > userBest ? '🎉 New personal best!' : `Personal best: ${localBest}`;
  document.getElementById('gameover-overlay').classList.add('active');
  if (score > 0) {
    try {
      const db   = await dbRead();
      const user = db.users[currentUser] || { hash:'', best:0, skin:'classic' };
      if (score > (user.best || 0)) {
        user.best = score; user.skin = selectedSkin;
        db.users[currentUser] = user;
        userBest = score;
        await dbWrite(db);
        toast('New best saved! 🎉', 'success');
        document.getElementById('menu-best').textContent = score;
        document.getElementById('hud-best').textContent  = score;
        localBest = score;
      }
    } catch(e) { console.error(e); }
  }
}

// ── CONTROLS ─────────────────────────────────────────────────────────────────
const KEY_MAP = {
  ArrowUp:'UP', ArrowDown:'DOWN', ArrowLeft:'LEFT', ArrowRight:'RIGHT',
  w:'UP', s:'DOWN', a:'LEFT', d:'RIGHT',
  W:'UP', S:'DOWN', A:'LEFT', D:'RIGHT',
};
document.addEventListener('keydown', e => {
  const action = KEY_MAP[e.key];
  if (!action) return;
  if (['ArrowUp','ArrowDown','ArrowLeft','ArrowRight'].includes(e.key)) e.preventDefault();
  if (!gameRunning) return;
  applyDir(action);
});
function dpadPress(a) { if (gameRunning) applyDir(a); }
function applyDir(a) {
  if (a==='UP'    && dir.y!== 1) nextDir = { x:0, y:-1 };
  if (a==='DOWN'  && dir.y!==-1) nextDir = { x:0, y: 1 };
  if (a==='LEFT'  && dir.x!== 1) nextDir = { x:-1, y:0 };
  if (a==='RIGHT' && dir.x!==-1) nextDir = { x: 1, y:0 };
}
let touchX, touchY;
document.addEventListener('touchstart', e => { touchX = e.touches[0].clientX; touchY = e.touches[0].clientY; }, { passive:true });
document.addEventListener('touchend', e => {
  if (!gameRunning) return;
  const dx = e.changedTouches[0].clientX - touchX;
  const dy = e.changedTouches[0].clientY - touchY;
  if (Math.abs(dx) > Math.abs(dy)) applyDir(dx > 0 ? 'RIGHT' : 'LEFT');
  else applyDir(dy > 0 ? 'DOWN' : 'UP');
}, { passive:true });
